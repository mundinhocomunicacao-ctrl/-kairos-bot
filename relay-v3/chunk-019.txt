etVoiceStatus('Ouvindo você…')};
  rec.onresult=e=>{
    const text=[...e.results].map(r=>r[0]?.transcript||'').join(' ').trim();
    if(gabiStudioLocalVoiceCommand(text))return;
    brief.value=text;brief.dispatchEvent(new Event('input',{bubbles:true}));
    gabiStudioSetVoiceStatus('Entendi. Criando a partir do que você falou…');gabiStudioGenerate('voice');
  };
  rec.onerror=()=>{gabiStudioSetPresenceState('idle');gabiStudioSetVoiceStatus('Não consegui ouvir. Tente novamente ou escreva o contexto.')};
  rec.onend=()=>{mic.classList.remove('listening');mic.textContent='🎙️ Falar';if(document.querySelector('[data-diva-3d]')?.dataset.state==='listening')gabiStudioSetPresenceState('idle');gabiStudioRecognition=null};
  rec.start();
}
