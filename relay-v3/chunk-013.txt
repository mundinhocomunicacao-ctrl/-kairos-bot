function contextCardHit(key,rows){
  const rule=contextCardRules[key]||contextCardRules.home;
  if(!rows.length)return null;
  if(!rule.terms.length)return rows[0]||null;
  return rows.find(row=>{
    const d=row.data||row;
    const hay=normalizedLiveText([d.category,d.status,d.title,d.summary].filter(Boolean).join(' '));
    return rule.terms.some(term=>hay.includes(normalizedLiveText(term)));
  })||null;
}
function showContextCard(key,sourceCard=null){
  const rule=contextCardRules[key]||contextCardRules.home;
  const rail=document.getElementById('contextCardRail');
  if(!rail)return false;
  const target=document.getElementById(rule.target)||document.getElementById('home');
  if(target&&rail.parentElement!==target)target.prepend(rail);
  rail.hidden=false;rail.dataset.contextSource=key;rail.dataset.contextLive='false';
  const cardTitle=sourceCard?.querySelector?.('.title')?.textContent?.trim();
  document.getElementById('contextCardTitle').textContent=cardTitle||rule.title;
  document.getElementById('contextNow').textContent=rule.now;
  document.getElementById('contextChanged').textContent=rule.changed;
  document.getElementById('contextWhy').textContent=rule.why;
  document.getElementById('contextNext').textContent=rule.next;
  document.getElementById('contextCardSource').textContent='base consolidada';
  const contextLive=document.getElementById('contextLive');
  contextLive.textContent='Base consolidada · aguardando novo movimento.';
  updateContextCardFromFeed(latestClientSafeRows);
  return true;
}
function updateContextCardFromFeed(rows){
  const rail=document.getElementById('contextCardRail');
  if(!rail||rail.hidden)return;
  const key=rail.dataset.contextSource||'home';
  const hit=contextCardHit(key,safeRadarRows(rows));
  const contextLive=document.getElementById('contextLive');
  const source=document.getElementById('contextCardSource');
  if(hit){
    const d=hit.data||hit;
    rail.dataset.contextLive='true';
    source.textContent='ao vivo';
    contextLive.textContent=['Novo movimento',feedText(d.status,70),feedText(d.title,160),feedText(d.summary,360),feedDate(d.occurredAt)].filter(Boolean).join(' · ');
  }else{
    rail.dataset.contextLive='false';
    source.textContent='base consolidada';
    contextLive.textContent='Base consolidada · nenhuma atualização nova altera esta leitura.';
  }
}

function openPanel(id){
  const p=document.getElementById(id);
  if(!p)return false;
  document.querySelectorAll('.tab,.panel').forEach(x=>x.classList.remove('active'));
  p.classList.add('active');
  const t=document.querySelector('.tab[data-tab="'+id+'"]');
  if(t)t.classList.add('active');
  requestAnimationFrame(()=>p.scrollIntoView({behavior:'smooth',block:'start'}));
  return true;
}
function handlePanelNavigation(e){
  const trigger=e.target?.closest?.('[data-open]');
  if(!trigger)return;
  const nested=e.target?.closest?.('a,button');
  if(nested&&nested!==trigger)return;
  const target=trigger.dataset.open;
  if(!target||!document.getElementById(target))return;
  e.preventDefault();
  showContextCard(target,trigger);
  openPanel(target);
}
function activatePanelNavigation(){
  document.querySelectorAll('[data-open]').forEach(el=>{
    el.setAttribute('tabindex','0');
    el.setAttribute('role','button');
    el.setAttribute('aria-controls',el.dataset.open);
  });
  document.addEventListener('click',handlePanelNavigation);
  document.addEventListener('keydown',e=>{
    const trigger=e.target?.closest?.('[data-open]');
    if(trigger&&(e.key==='Enter'||e.key===' ')){
      e.preventDefault();
      if(document.getElementById(trigger.dataset.open)){showContextCard(trigger.dataset.open,trigger);openPanel(trigger.dataset.open);}
    }
  });
}
