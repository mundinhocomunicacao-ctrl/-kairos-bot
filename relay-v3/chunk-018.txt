async function gabiStudioGenerate(intent=gabiStudioIntent,overrideBrief=''){
  const composer=document.getElementById('gabiStudioComposer'),button=document.getElementById('gabiStudioGenerate'),brief=(overrideBrief||document.getElementById('gabiStudioBrief').value).trim();
  gabiStudioLastBrief=brief;
  if(gabiStudioLocalVoiceCommand(brief))return;
  const preset=gabiStudioSmartPreset(intent,brief);composer.classList.add('studioLoading');button.textContent='Criando…';gabiStudioSetPresenceState('thinking');gabiStudioSetVoiceStatus('DIVA está organizando a melhor direção…');
  try{
    const res=await fetch(GABI_DIVA_STUDIO_ENDPOINT,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({intent,brief,preset,context:gabiStudioPublicContext()})});
    const data=await res.json().catch(()=>({}));
    if(!res.ok)throw new Error(data.message||'Não consegui criar agora.');
    renderGabiStudioResult(data);
  }catch(error){
    renderGabiStudioResult(gabiStudioLocalFallback(intent,brief,gabiStudioPublicContext()));
    gabiStudioSetVoiceStatus('Tenho uma alternativa pronta para você.');
  }finally{composer.classList.remove('studioLoading');if(document.querySelector('[data-diva-3d]')?.dataset.state==='thinking')gabiStudioSetPresenceState('idle');button.textContent='Criar →'}
}

}
