function activateLiveCharts(){
  if(!document.getElementById('liveChartTip')){
    const tip=document.createElement('div');tip.id='liveChartTip';tip.className='liveChartTip';tip.setAttribute('role','status');document.body.appendChild(tip);
  }
  const tip=document.getElementById('liveChartTip');
  const charts=[...document.querySelectorAll('.donut,.barrow,.ibiCol,.flow>div,.metricBox,.track')];
  const observer='IntersectionObserver' in window?new IntersectionObserver(entries=>entries.forEach(entry=>{if(entry.isIntersecting)entry.target.classList.add('isSeen')}),{threshold:.18}):null;
  charts.forEach(el=>{
    if(el.dataset.liveChartReady==='true')return;
    el.dataset.liveChartReady='true';el.classList.add('liveChart');
    if(el.matches('.barrow,.ibiCol,.track'))el.dataset.chartMotion='bar';
    el.setAttribute('tabindex','0');el.setAttribute('role','group');
    const label=feedText(el.innerText||el.closest('.card')?.querySelector('.title')?.textContent||'Gráfico do Radar',220);
    el.setAttribute('aria-label',label);
    const show=e=>{
      tip.textContent=label+' · '+(el.closest('.card')?.dataset.liveState==='updated'?'há novo movimento':'base consolidada');
      tip.style.left=Math.min(window.innerWidth-280,(e.clientX||el.getBoundingClientRect().left)+12)+'px';
      tip.style.top=Math.min(window.innerHeight-80,(e.clientY||el.getBoundingClientRect().bottom)+10)+'px';
      tip.classList.add('show');
    };
    const hide=()=>tip.classList.remove('show');
    el.addEventListener('mousemove',show);el.addEventListener('mouseenter',show);el.addEventListener('mouseleave',hide);
    el.addEventListener('focus',()=>show({clientX:el.getBoundingClientRect().left,clientY:el.getBoundingClientRect().bottom}));el.addEventListener('blur',hide);
    if(observer)observer.observe(el);else el.classList.add('isSeen');
  });
  document.querySelectorAll('.card[data-live-surface="true"]').forEach(card=>{
    if(card.querySelector(':scope > .chartLiveBadge'))return;
    if(card.querySelector('.donut,.barrow,.ibiChart,.flow,.metricBox,.track')){
      const badge=document.createElement('div');badge.className='chartLiveBadge';badge.textContent='gráfico vivo · base consolidada';
      const firstChart=card.querySelector('.donutWrap,.ibiChart,.flow,.barrow,.metricBox,.track');
      if(firstChart)firstChart.insertAdjacentElement('afterend',badge);
    }
  });
}

async function visitorToken(){
  const cached=localStorage.getItem(GABI_RADAR_TOKEN_KEY);
  const exp=Number(localStorage.getItem(GABI_RADAR_TOKEN_EXP_KEY)||0);
  if(cached&&Date.now()<exp-60000)return cached;
  const res=await fetch('https://www.wixapis.com/oauth2/token',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({clientId:GABI_RADAR_CLIENT_ID,grantType:'anonymous'})});
  if(!res.ok)throw new Error('visitor_token_'+res.status);
  const data=await res.json();
  if(!data.access_token)throw new Error('visitor_token_missing');
  localStorage.setItem(GABI_RADAR_TOKEN_KEY,data.access_token);
  localStorage.setItem(GABI_RADAR_TOKEN_EXP_KEY,String(Date.now()+Math.max(300,Number(data.expires_in||14400))*1000));
  return data.access_token;
}

async function loadLiveFeed(){
  const status=document.getElementById('feedStatus');
  try{
    const token=await visitorToken();
    const res=await fetch('https://www.wixapis.com/wix-data/v2/items/query',{method:'POST',headers:{'content-type':'application/json','authorization':token},body:JSON.stringify({dataCollectionId:GABI_RADAR_COLLECTION_ID,query:{paging:{limit:100,offset:0}}})});
    if(!res.ok)throw new Error('feed_query_'+res.status);
    const data=await res.json();
    const rows=(Array.isArray(data.dataItems)?data.dataItems:[]).filter(x=>x?.data?.clientSafe===true&&!['sistema','system','bootstrap'].includes(String(x?.data?.category||'').toLowerCase())).sort((a,b)=>new Date(b?.data?.occurredAt||0)-new Date(a?.data?.occurredAt||0)).slice(0,6);
    renderLiveFeed(rows);
  }catch(error){
    console.error('GABI_RADAR_LIVE_FEED',error);
    status.textContent='leitura consolidada';status.classList.add('error');
    const grid=document.getElementById('liveFeedGrid');grid.replaceChildren();
    const empty=document.createElement('div');empty.className='liveFeedEmpty';empty.textContent='A atualização ao vivo está temporariamente indisponível. O restante do painel continua usando a última leitura reconciliada.';grid.appendChild(empty);
  }
}
