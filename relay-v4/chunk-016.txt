    const key=normalizedLiveText(d.title);

function safeRadarRows(rows){
  return (Array.isArray(rows)?rows:[]).filter(x=>x?.data?.clientSafe===true&&!['sistema','system','bootstrap'].includes(String(x?.data?.category||'').toLowerCase()));
}
function liveRowForCard(card,rows){
  const cardText=normalizedLiveText(card.innerText+' '+(card.dataset.open||'')+' '+(card.dataset.stage||''));
  const rule=LIVE_CARD_RULES.find(r=>r.terms.some(t=>cardText.includes(normalizedLiveText(t))));
  if(!rule)return null;
  return rows.find(row=>{
    const d=row.data||row;
    const hay=normalizedLiveText([d.category,d.status,d.title,d.summary].filter(Boolean).join(' '));
    return rule.categories.some(c=>hay.includes(normalizedLiveText(c)));
  })||null;
}
function ensureLiveCardChrome(card){
  let meta=card.querySelector(':scope > .liveCardMeta');
  if(!meta){meta=document.createElement('div');meta.className='liveCardMeta';card.prepend(meta)}
  let signal=card.querySelector(':scope > .liveCardSignal');
  if(!signal){signal=document.createElement('div');signal.className='liveCardSignal';card.appendChild(signal)}
  return{meta,signal};
}
function hydrateLiveCards(rows){
  const safeRows=safeRadarRows(rows);
  document.querySelectorAll('.card[data-live-surface="true"]').forEach(card=>{
    const {meta,signal}=ensureLiveCardChrome(card);
    const hit=liveRowForCard(card,safeRows);
    if(hit){
      const d=hit.data||hit;
      card.dataset.liveState='updated';
      meta.textContent=['ao vivo',feedText(d.status,60),feedDate(d.occurredAt)].filter(Boolean).join(' · ');
      signal.textContent='Último movimento: '+(feedText(d.summary,420)||feedText(d.title,180)||'Atualização reconciliada.');
    }else{
      card.dataset.liveState='consolidated';
      meta.textContent='base consolidada · aguardando novo movimento';
      signal.textContent='';
    }
  });
  document.querySelectorAll('.chartLiveBadge').forEach(badge=>{
    const card=badge.closest('.card');
    badge.textContent=card?.dataset.liveState==='updated'?'gráfico vivo · novo movimento no Radar':'gráfico vivo · base consolidada';
  });
}

  });
  document.querySelectorAll('.card[data-live-surface="true"]').forEach(card=>{
    if(card.querySelector(':scope > .chartLiveBadge'))return;

  });
