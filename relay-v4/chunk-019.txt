async function gabiStudioGenerate(intent=gabiStudioIntent,overrideBrief=''){
  const composer=document.getElementById('gabiStudioComposer'),button=document.getElementById('gabiStudioGenerate'),brief=(overrideBrief||document.getElementById('gabiStudioBrief').value).trim();
  gabiStudioLastBrief=brief;
  if(gabiStudioLocalVoiceCommand(brief))return;
  const preset=gabiStudioSmartPreset(intent,brief);composer.classList.add('studioLoading');button.textContent='Criando…';gabiStudioSetPresenceState('thinking');gabiStudioSetVoiceStatus('DIVA está organizando a melhor direção…');
  try{
    const res=await fetch(GABI_DIVA_STUDIO_ENDPOINT,{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({intent,brief,preset,context:gabiStudioPublicContext()})});
    const data=await res.json().catch(()=>({}));
    if(!res.ok)throw new Error(data.message||'Não consegui criar agora.');
    renderGabiStudioResult(data);
  }catch(error){
    renderGabiStudioResult(gabiStudioLocalFallback(intent,brief,gabiStudioPublicContext()));
    gabiStudioSetVoiceStatus('Tenho uma alternativa pronta para você.');
  }finally{composer.classList.remove('studioLoading');if(document.querySelector('[data-diva-3d]')?.dataset.state==='thinking')gabiStudioSetPresenceState('idle');button.textContent='Criar →'}
}
function gabiStudioRefine(kind){
  const map={
    another:'Crie uma direção completamente diferente da anterior, mantendo o mesmo objetivo.',
    shorter:'Faça uma versão mais curta, direta e fácil de gravar.',
    funny:'Deixe a ideia mais leve, espontânea e divertida, sem forçar humor.',
    caption:'Priorize uma legenda forte e curta para acompanhar a ideia.'
  };
  const instruction=map[kind];if(!instruction)return;
  const previous=gabiStudioLastResult?.idea?('Ideia anterior: '+gabiStudioLastResult.idea):'';
  const base=document.getElementById('gabiStudioBrief')?.value.trim()||gabiStudioLastBrief||'';
  gabiStudioGenerate(gabiStudioIntent,[base,previous,instruction].filter(Boolean).join('\n'));
}
function startGabiStudioMic(){
  const Recognition=window.SpeechRecognition||window.webkitSpeechRecognition;
  if(!Recognition){gabiStudioSetVoiceStatus('Este navegador não liberou reconhecimento de voz. Você ainda pode escrever o briefing.');return}
  if(gabiStudioRecognition){try{gabiStudioRecognition.stop()}catch{}}
  const mic=document.getElementById('gabiStudioMic'),brief=document.getElementById('gabiStudioBrief');
  const rec=new Recognition();gabiStudioRecognition=rec;rec.lang='pt-BR';rec.interimResults=false;rec.continuous=false;
  rec.onstart=()=>{mic.classList.add('listening');mic.textContent='■ Ouvindo';gabiStudioSetPresenceState('listening');gabiStudioSetVoiceStatus('Ouvindo você…')};
  rec.onresult=e=>{
    const text=[...e.results].map(r=>r[0]?.transcript||'').join(' ').trim();
    if(gabiStudioLocalVoiceCommand(text))return;
    brief.value=text;brief.dispatchEvent(new Event('input',{bubbles:true}));
    gabiStudioSetVoiceStatus('Entendi. Criando a partir do que você falou…');gabiStudioGenerate('voice');
  };
  rec.onerror=()=>{gabiStudioSetPresenceState('idle');gabiStudioSetVoiceStatus('Não consegui ouvir. Tente novamente ou escreva o contexto.')};
  rec.onend=()=>{mic.classList.remove('listening');mic.textContent='🎙️ Falar';if(document.querySelector('[data-diva-3d]')?.dataset.state==='listening')gabiStudioSetPresenceState('idle');gabiStudioRecognition=null};
  rec.start();
}
