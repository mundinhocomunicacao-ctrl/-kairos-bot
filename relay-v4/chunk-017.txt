    if(!res.ok)throw new Error('feed_query_'+res.status);
    const data=await res.json();
    const rows=(Array.isArray(data.dataItems)?data.dataItems:[]).filter(x=>x?.data?.clientSafe===true&&!['sistema','system','bootstrap'].includes(String(x?.data?.category||'').toLowerCase())).sort((a,b)=>new Date(b?.data?.occurredAt||0)-new Date(a?.data?.occurredAt||0)).slice(0,6);

    console.error('GABI_RADAR_LIVE_FEED',error);
    status.textContent='leitura consolidada';status.classList.add('error');

  }
}

function gabiStudioSmartPreset(intent=gabiStudioIntent,brief=''){
  const text=normalizedLiveText(brief||'');
  const minutes=Number((text.match(/(\d{1,3})\s*(?:min|minuto|minutos)/)||[])[1]||0);
  const fast=intent==='record_now'||(minutes&&minutes<=20)||/agora|rápido|rapidinho|correndo/.test(text);
  const reference=intent==='reference'||/refer[eê]ncia|trend|tend[eê]ncia|vídeo que vi/.test(text);
  const intimate=/desabafo|pessoal|sincero|sincera|vulner[aá]vel|reflex/.test(text);
  const playful=/engra[cç]|leve|divertid|humor|meme/.test(text);
  const location=/carro/.test(text)?'carro':/faculdade|aula/.test(text)?'faculdade':/aeroporto|avi[aã]o|viagem/.test(text)?'viagem':/casa|quarto|sala/.test(text)?'casa':'rotina';
  const duration=fast?'15–25s':intimate?'35–50s':'25–35s';
  const tone=playful?'leve e espontâneo':intimate?'íntimo e direto':reference?'original com linguagem de referência':'natural e pessoal';
  const production=fast?'1–3 planos · edição mínima':location==='rotina'?'3–5 planos · produção leve':`3–5 planos · aproveitar ${location}`;
  const format=reference?'Reels/TikTok reinterpretado':'Reels/TikTok vertical';
  const preset={format,duration,tone,production,location,fast,reference};
  const summary=document.getElementById('gabiStudioPresetSummary');if(summary)summary.textContent=`${format} · ${duration} · ${tone} · ${production}`;
  return preset;
}

function gabiStudioPickVoice(voices=[]){
  const score=v=>{
    const name=String(v?.name||'').toLowerCase(),lang=String(v?.lang||'').toLowerCase();
    let s=0;if(lang==='pt-br')s+=100;else if(lang.startsWith('pt'))s+=55;else return-999;
    if(/google.*portugu|microsoft.*francisca|luciana|camila|fernanda|leticia|maria/.test(name))s+=34;
    if(v?.localService)s+=8;if(/natural|neural|premium|enhanced/.test(name))s+=18;if(/compact|espeak/.test(name))s-=30;
    return s;
  };
  return [...voices].sort((a,b)=>score(b)-score(a))[0]||null;
}
function gabiStudioSpeak(text){
  gabiStudioLastSpeech=gabiStudioPublicText(text,2600);
  if(!gabiStudioVoiceEnabled||!gabiStudioLastSpeech||!('speechSynthesis' in window))return;
  window.speechSynthesis.cancel();
  const spoken=gabiStudioLastSpeech.replace(/[`*_#>]/g,' ').replace(/\s+/g,' ').trim();
  const u=new SpeechSynthesisUtterance(spoken);
  u.lang='pt-BR';u.rate=.91;u.pitch=.98;u.volume=.96;
  const preferred=gabiStudioPickVoice(window.speechSynthesis.getVoices());
  if(preferred)u.voice=preferred;
  u.onstart=()=>{gabiStudioSetPresenceState('speaking');gabiStudioSetVoiceStatus('DIVA falando… diga “pare” para interromper.')};
  u.onend=()=>{gabiStudioSetPresenceState('idle');gabiStudioSetVoiceStatus('Pronta para continuar.')};
  u.onerror=()=>{gabiStudioSetPresenceState('idle');gabiStudioSetVoiceStatus('A voz falhou, mas a resposta continua aqui na tela.')};
  window.speechSynthesis.speak(u);
}
